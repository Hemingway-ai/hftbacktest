### 2.2 Numba JIT 编译

策略函数必须使用 `@njit` 装饰器编译：

```python
from numba import njit

@njit
def strategy(hbt):
    # 这里的代码会被编译为机器码
    # 注意：不是所有 Python 语法都支持
    pass
```

**Numba 限制**：
- 不支持 Python 类
- 不支持字典 (使用 `numba.typed.Dict`)
- 不支持部分 NumPy 函数
- 需要显式类型标注

**推荐的 Numba 模式**：

```python
from numba import njit, uint64
from numba.typed import Dict

@njit
def strategy(hbt):
    # 使用 numba.typed.Dict 替代普通字典
    orders = Dict.empty(uint64, float64)

    # 使用 numpy 函数
    mid_price = np.mean([depth.best_bid, depth.best_ask])

    # 循环和条件
    for i in range(10):
        if condition:
            pass
```

## 三、核心模式

### 3.1 做市策略基本模式

做市策略的核心是在 bid/ask 两侧挂单，赚取价差：

```python
@njit
def market_making(hbt, half_spread, order_qty):
    asset_no = 0

    while hbt.elapse(1_000_000_000) == 0:
        hbt.clear_inactive_orders(asset_no)

        depth = hbt.depth(asset_no)
        position = hbt.position(asset_no)
        orders = hbt.orders(asset_no)

        mid_price = (depth.best_bid + depth.best_ask) / 2.0

        # 计算报价
        bid_price = mid_price - half_spread
        ask_price = mid_price + half_spread

        # 生成订单 ID (使用价格 tick)
        bid_id = uint64(round(bid_price / depth.tick_size))
        ask_id = uint64(round(ask_price / depth.tick_size)) + 1_000_000

        # 提交订单 (如果不存在)
        if bid_id not in orders:
            hbt.submit_buy_order(
                asset_no, bid_id, bid_price,
                order_qty, GTX, LIMIT, False
            )

        if ask_id not in orders:
            hbt.submit_sell_order(
                asset_no, ask_id, ask_price,
                order_qty, GTX, LIMIT, False
            )

        # 等待订单响应
        if not hbt.wait_order_response(asset_no, -1, 5_000_000_000):
            return False

    return True
```

### 3.2 库存管理模式

根据持仓调整报价，控制风险：

```python
@njit
def market_making_with_skew(hbt, half_spread, skew, order_qty, max_position):
    asset_no = 0

    while hbt.elapse(1_000_000_000) == 0:
        hbt.clear_inactive_orders(asset_no)

        depth = hbt.depth(asset_no)
        position = hbt.position(asset_no)
        orders = hbt.orders(asset_no)

        mid_price = (depth.best_bid + depth.best_ask) / 2.0

        # 库存偏斜: 多头时降低报价，空头时提高报价
        normalized_pos = position / order_qty
        reservation_price = mid_price - skew * normalized_pos

        # 计算报价
        bid_price = reservation_price - half_spread
        ask_price = reservation_price + half_spread

        # 检查持仓限制
        can_buy = position < max_position
        can_sell = position > -max_position

        bid_id = uint64(round(bid_price / depth.tick_size))
        ask_id = uint64(round(ask_price / depth.tick_size)) + 1_000_000

        if can_buy and bid_id not in orders:
            hbt.submit_buy_order(
                asset_no, bid_id, bid_price,
                order_qty, GTX, LIMIT, False
            )

        if can_sell and ask_id not in orders:
            hbt.submit_sell_order(
                asset_no, ask_id, ask_price,
                order_qty, GTX, LIMIT, False
            )

        if not hbt.wait_order_response(asset_no, -1, 5_000_000_000):
            return False

    return True
```

### 3.3 Alpha 信号模式

基于市场信号调整报价：

```python
@njit
def alpha_market_making(hbt, half_spread, alpha_coef, order_qty):
    asset_no = 0
    alpha_history = np.full(10000, np.nan)
    t = 0

    while hbt.elapse(1_000_000_000) == 0:
        depth = hbt.depth(asset_no)

        # 计算 alpha 信号 (例如: 订单簿失衡)
        bid_qty = depth.best_bid_qty
        ask_qty = depth.best_ask_qty
        alpha = (bid_qty - ask_qty) / (bid_qty + ask_qty)

        # 标准化 alpha
        alpha_history[t] = alpha
        window = min(t + 1, 1000)
        m = np.nanmean(alpha_history[max(0, t-window):t+1])
        s = np.nanstd(alpha_history[max(0, t-window):t+1])
        z_score = (alpha - m) / s if s > 0 else 0

        # 调整公平价格
        mid_price = (depth.best_bid + depth.best_ask) / 2.0
        fair_price = mid_price + alpha_coef * z_score

        # 基于 fair_price 报价
        bid_price = fair_price - half_spread
        ask_price = fair_price + half_spread

        # ... 提交订单逻辑

        t += 1
```

### 3.4 网格交易模式

在多个价格档位挂单：

```python
@njit
def grid_trading(hbt, grid_num, grid_interval, order_qty):
    asset_no = 0

    while hbt.elapse(1_000_000_000) == 0:
        hbt.clear_inactive_orders(asset_no)

        depth = hbt.depth(asset_no)
        mid_price = (depth.best_bid + depth.best_ask) / 2.0
        orders = hbt.orders(asset_no)

        # 生成网格订单
        new_orders = Dict.empty(uint64, float64)

        for i in range(grid_num):
            # 买单网格
            bid_price = depth.best_bid - i * grid_interval
            bid_id = uint64(round(bid_price / depth.tick_size))
            new_orders[bid_id] = bid_price

            # 卖单网格
            ask_price = depth.best_ask + i * grid_interval
            ask_id = uint64(round(ask_price / depth.tick_size)) + 1_000_000
            new_orders[ask_id] = ask_price

        # 取消不在新网格中的订单
        order_values = orders.values()
        while order_values.has_next():
            order = order_values.get()
            if order.cancellable and order.order_id not in new_orders:
                hbt.cancel(asset_no, order.order_id, False)

        # 提交新订单
        for order_id, price in new_orders.items():
            if order_id not in orders:
                if order_id < 1_000_000:
                    hbt.submit_buy_order(
                        asset_no, order_id, price,
                        order_qty, GTX, LIMIT, False
                    )
                else:
                    hbt.submit_sell_order(
                        asset_no, order_id, price,
                        order_qty, GTX, LIMIT, False
                    )
```

## 四、订单管理

### 4.1 订单 ID 设计

推荐使用价格 tick 作为订单 ID：

```python
# 买单 ID: 价格 tick
bid_id = uint64(round(bid_price / tick_size))

# 卖单 ID: 价格 tick + 偏移量
ask_id = uint64(round(ask_price / tick_size)) + 1_000_000_000
```

**优点**：
- ID 唯一性由价格保证
- 便于调试和追踪
- 便于检查订单是否存在

### 4.2 订单生命周期

```
提交订单
    │
    ▼
订单进入队列 (status: NEW)
    │
    ├──► 被成交 (status: FILLED)
    │
    ├──► 被取消 (status: CANCELED)
    │
    └──► 部分成交 (status: PARTIALLY_FILLED)
            │
            └──► 继续成交或取消
```

### 4.3 订单清理

定期清理失效订单：


### 4.4 订单取消

## 八、完整示例

### 8.1 简单做市策略

```python
import numpy as np
from numba import njit, uint64
from hftbacktest import (
    BacktestAsset,
    ROIVectorMarketDepthBacktest,
    GTX, LIMIT, BUY, SELL,
    Recorder,
)
from hftbacktest.stats import LinearAssetRecord

@njit
def simple_market_maker(hbt, stat, half_spread, order_qty):
    asset_no = 0

    while hbt.elapse(1_000_000_000) == 0:
        hbt.clear_inactive_orders(asset_no)

        depth = hbt.depth(asset_no)
        position = hbt.position(asset_no)
        orders = hbt.orders(asset_no)

        mid_price = (depth.best_bid + depth.best_ask) / 2.0

        bid_price = mid_price - half_spread
        ask_price = mid_price + half_spread

        bid_id = uint64(round(bid_price / depth.tick_size))
        ask_id = uint64(round(ask_price / depth.tick_size)) + 1_000_000

        if bid_id not in orders:
            hbt.submit_buy_order(
                asset_no, bid_id, bid_price,
                order_qty, GTX, LIMIT, False
            )

        if ask_id not in orders:
            hbt.submit_sell_order(
                asset_no, ask_id, ask_price,
                order_qty, GTX, LIMIT, False
            )

        if not hbt.wait_order_response(asset_no, -1, 5_000_000_000):
            return False

        stat.record(hbt)

    return True

# 配置
roi_lb = 0
roi_ub = 3000

latency_data = np.load('feed_latency.npz')['data']

asset = (
    BacktestAsset()
    .data(['market_data.npz'])
    .initial_snapshot('snapshot.npz')
    .linear_asset(1.0)
    .intp_order_latency(latency_data)
    .power_prob_queue_model(3)
    .no_partial_fill_exchange()
    .trading_value_fee_model(-0.00005, 0.0007)
    .tick_size(0.1)
    .lot_size(0.001)
    .roi_lb(roi_lb)
    .roi_ub(roi_ub)
)

hbt = ROIVectorMarketDepthBacktest([asset])
stat = Recorder()

simple_market_maker(hbt, stat, half_spread=1.0, order_qty=1.0)
hbt.close()

# 分析结果
data = np.load('result.npz')['0']
stats = LinearAssetRecord(data).resample('5m').stats(book_size=1_000_000)
stats.summary()
```

## 九、最佳实践总结

| 实践 | 说明 |
|------|------|
| 使用 `@njit` | 所有策略函数必须 JIT 编译 |
| 清理失效订单 | 每个循环开始调用 `clear_inactive_orders` |
| 合理设置订单 ID | 使用价格 tick 作为 ID |
| 控制持仓风险 | 设置最大持仓限制 |
| 使用 GTX | 做市策略使用 GTX 确保只做 Maker |
| 等待订单响应 | 使用 `wait_order_response` 确保订单处理完成 |
| 测试不同参数 | 使用网格搜索优化参数 |
| 监控延迟 | 关注 `order_latency` 返回值 |
| 使用 ROI | 价格范围可预测时使用 ROIVector |
| 记录状态 | 使用 Recorder 记录回测过程 |
